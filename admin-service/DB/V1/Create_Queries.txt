 created_at TIMESTAMP NOT NULL,

  CREATE TABLE flight_schedule (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    flight_id BIGINT NOT NULL,

    source_airport_id BIGINT NOT NULL,
    destination_airport_id BIGINT NOT NULL,

    departure_time TIME NOT NULL,
    arrival_time TIME NOT NULL,

    arrival_day_offset TINYINT NOT NULL DEFAULT 0,

    effective_from DATE NOT NULL,
    effective_to DATE NOT NULL,

    created_by VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL,

    modified_by VARCHAR(255) NOT NULL,
    modified_at TIMESTAMP NOT NULL,

    CONSTRAINT chk_source_destination
        CHECK (source_airport_id <> destination_airport_id),

    CONSTRAINT chk_effective_dates
        CHECK (effective_to >= effective_from),

    CONSTRAINT chk_arrival_day_offset
        CHECK (arrival_day_offset IN (0, 1)),

    CONSTRAINT chk_arrival_departure_time
        CHECK (
            (arrival_day_offset = 0 AND arrival_time > departure_time)
            OR
            (arrival_day_offset = 1 AND arrival_time < departure_time)
        ),

    CONSTRAINT fk_flight_schedule_flight
        FOREIGN KEY (flight_id)
        REFERENCES flight (id),

    CONSTRAINT fk_flight_schedule_source_airport
        FOREIGN KEY (source_airport_id)
        REFERENCES airport (id),

    CONSTRAINT fk_flight_schedule_destination_airport
        FOREIGN KEY (destination_airport_id)
        REFERENCES airport (id),

    CONSTRAINT uk_flight_schedule_unique
        UNIQUE (
            flight_id,
            departure_time,
            effective_from
        )
);

CREATE TABLE flight_schedule_instance (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    flight_schedule_id BIGINT NOT NULL,

    travel_date DATE NOT NULL,

    actual_departure_time TIME,

    actual_arrival_time TIME,

    status VARCHAR(50) NOT NULL,

    gate VARCHAR(100),

    terminal VARCHAR(100),

    created_by VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL,

    modified_by VARCHAR(255) NOT NULL,
    modified_at TIMESTAMP NOT NULL,

    CONSTRAINT fk_flight_schedule_instance_schedule
        FOREIGN KEY (flight_schedule_id)
        REFERENCES flight_schedule(id),

    CONSTRAINT uk_flight_schedule_instance
        UNIQUE (flight_schedule_id, travel_date),

    CONSTRAINT chk_flight_schedule_instance_status
        CHECK (
            status IN (
                'SCHEDULED',
                'DELAYED',
                'BOARDING',
                'DEPARTED',
                'ARRIVED',
                'CANCELLED'
            )
        )
);

CREATE TABLE flight_seat_availability (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    flight_schedule_instance_id BIGINT NOT NULL,

    seat_class_id BIGINT NOT NULL,

    total_seats INT NOT NULL,

    booked_seats INT NOT NULL DEFAULT 0,

    price DECIMAL(10,2) NOT NULL,

    created_by VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL,

    CONSTRAINT chk_total_seats
        CHECK (total_seats > 0),

    CONSTRAINT chk_booked_seats
        CHECK (booked_seats >= 0),

    CONSTRAINT chk_booked_less_than_total
        CHECK (booked_seats <= total_seats),

    CONSTRAINT chk_price
        CHECK (price > 0),

    CONSTRAINT fk_seat_availability_schedule_instance
        FOREIGN KEY (flight_schedule_instance_id)
        REFERENCES flight_schedule_instance(id),

    CONSTRAINT fk_seat_availability_seat_class
        FOREIGN KEY (seat_class_id)
        REFERENCES seat_class(id),

    CONSTRAINT uk_seat_availability
        UNIQUE (
            flight_schedule_instance_id,
            seat_class_id
        )
);

CREATE TABLE flight_seat_configuration (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    flight_id BIGINT NOT NULL,

    seat_class_id BIGINT NOT NULL,

    total_seats INT NOT NULL,

    created_by VARCHAR(255) NOT NULL,

    created_at TIMESTAMP NOT NULL,

    modified_by VARCHAR(255) NOT NULL,

    modified_at TIMESTAMP NOT NULL,

    CONSTRAINT flight_seat_config_chk_total_seats
        CHECK (total_seats > 0),

    CONSTRAINT fk_flight_seat_configuration_flight
        FOREIGN KEY (flight_id)
        REFERENCES flight(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    CONSTRAINT fk_flight_seat_configuration_seat_class
        FOREIGN KEY (seat_class_id)
        REFERENCES seat_class(id)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    CONSTRAINT uk_flight_seat_configuration
        UNIQUE (
            flight_id,
            seat_class_id
        )
);

CREATE TABLE booking (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    booking_reference VARCHAR(20) NOT NULL,

    user_id BIGINT NOT NULL,

    flight_schedule_instance_id BIGINT NOT NULL,

    seat_class_id BIGINT NOT NULL,

    no_of_seats INT NOT NULL,

    price_per_seat DECIMAL(10,2) NOT NULL,

    total_amount DECIMAL(10,2) NOT NULL,

    booking_status VARCHAR(30) NOT NULL,

    payment_status VARCHAR(30) NOT NULL,

    created_by VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL,

    modified_by VARCHAR(255) NOT NULL,
    modified_at TIMESTAMP NOT NULL,

    CONSTRAINT uk_booking_reference
        UNIQUE (booking_reference),

    CONSTRAINT chk_booking_no_of_seats
        CHECK (no_of_seats > 0),

    CONSTRAINT chk_booking_price_per_seat
        CHECK (price_per_seat > 0),

    CONSTRAINT chk_booking_total_amount
        CHECK (total_amount > 0),

    CONSTRAINT fk_booking_user
        FOREIGN KEY (user_id)
        REFERENCES user_account(id),

    CONSTRAINT fk_booking_flight_instance
        FOREIGN KEY (flight_schedule_instance_id)
        REFERENCES flight_schedule_instance(id),

    CONSTRAINT fk_booking_seat_class
        FOREIGN KEY (seat_class_id)
        REFERENCES seat_class(id)
);

CREATE TABLE booking_passenger (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    booking_id BIGINT NOT NULL,

    first_name VARCHAR(100) NOT NULL,

    last_name VARCHAR(100),

    age int NOT NULL,

    gender VARCHAR(20) NOT NULL,

    passport_number VARCHAR(50),

    created_by VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL,

    modified_by VARCHAR(255) NOT NULL,
    modified_at TIMESTAMP NOT NULL,

    CONSTRAINT fk_booking_passenger_booking
        FOREIGN KEY (booking_id)
        REFERENCES booking(id)
        ON DELETE CASCADE,

    CONSTRAINT chk_passenger_first_name
        CHECK (TRIM(first_name) <> ''),

    CONSTRAINT chk_passenger_gender
        CHECK (TRIM(gender) <> '')
);
